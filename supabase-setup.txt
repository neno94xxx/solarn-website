-- SOLARNI PORTAL — POKRETANJE U NOVOM SUPABASE PROJEKTU
-- Ova datoteka je u cijelosti valjan SQL (upute su SQL komentari).
-- 1. Kreirajte NOVI projekt na https://supabase.com/dashboard.
-- 2. Authentication > Users > Add user > Create new user:
--    e-mail: ferdinand.nodilo@yahoo.com
--    odaberite svoju jaku lozinku i ukljucite Auto Confirm User.
--    Lozinka se ne upisuje u ovu datoteku niti u kod aplikacije.
-- 3. U SQL Editor zalijepite CIJELU ovu datoteku i kliknite Run.
-- 4. U Settings > API/Data API preuzmite Project URL i publishable key.
--    Kopirajte .env.example u .env u projektu Solarni portal i popunite:
--    SUPABASE_URL=https://VAS-PROJEKT.supabase.co
--    SUPABASE_PUBLISHABLE_KEY=sb_publishable_...
--    Za stariji anon JWT koristite SUPABASE_ANON_KEY umjesto publishable key.
--    NEMOJTE koristiti secret ili service_role kljuc.
-- 5. Authentication > URL Configuration:
--    tijekom razvoja Site URL: http://localhost:4321
--    prije objave postavite stvarnu HTTPS domenu (i SITE_URL u .env).
--    Ne morate ukljuciti javnu registraciju; korisnike dodaje vlasnik projekta.
-- 6. Pokrenite npm run dev i otvorite http://localhost:4321/admin/.
--
-- Kreira samo solar_* tablice/funkcije/politike i solar-articles bucket.
-- Slike se nalaze u javnom Storage bucketu (ukljucujuci spremljene nacrte),
-- a podaci clanaka u Postgresu. API prije spremanja optimizira slike u WebP.
-- Nacrti clanaka nisu javno citljivi. Jedino upis u solar_admins dodjeljuje
-- pravo uredivanja; obican Supabase racun sam po sebi nema te ovlasti.
-- Skripta ne kreira laznu autentifikaciju ni lozinke u vlastitim tablicama.

begin;

create table if not exists public.solar_admins (
  user_id uuid primary key references auth.users(id) on delete cascade,
  created_at timestamptz not null default now()
);
alter table public.solar_admins enable row level security;
revoke all on public.solar_admins from anon, authenticated;
grant select on public.solar_admins to authenticated;
drop policy if exists solar_admin_read_self on public.solar_admins;
create policy solar_admin_read_self on public.solar_admins for select to authenticated using (user_id = (select auth.uid()));

create or replace function public.is_solar_admin()
returns boolean language sql stable security definer set search_path = ''
as $$ select exists(select 1 from public.solar_admins where user_id = (select auth.uid())); $$;
revoke all on function public.is_solar_admin() from public;
grant execute on function public.is_solar_admin() to anon, authenticated;

create table if not exists public.solar_media (
  id uuid primary key default gen_random_uuid(),
  owner_id uuid not null references auth.users(id),
  variants jsonb not null check (jsonb_typeof(variants) = 'array' and jsonb_array_length(variants) between 1 and 3),
  width integer not null check (width between 1 and 1600),
  height integer not null check (height between 1 and 30000),
  original_name text not null check (char_length(original_name) <= 255),
  created_at timestamptz not null default now()
);

create or replace function public.solar_valid_blocks(blocks jsonb)
returns boolean language plpgsql immutable set search_path = '' as $$
declare block jsonb;
begin
  if jsonb_typeof(blocks) <> 'array' or jsonb_array_length(blocks) > 80 then return false; end if;
  for block in select * from jsonb_array_elements(blocks) loop
    if jsonb_typeof(block) <> 'object' or coalesce(block->>'id','') !~* '^[0-9a-f]{8}-[0-9a-f]{4}-[1-5][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$' then return false; end if;
    if block->>'type' in ('heading','text') then
      if jsonb_typeof(block->'text') is distinct from 'string' or char_length(trim(block->>'text')) < 1 or char_length(block->>'text') > (case when block->>'type' = 'heading' then 220 else 20000 end) then return false; end if;
    elsif block->>'type' = 'image' then
      if coalesce(block->>'mediaId','') !~* '^[0-9a-f]{8}-[0-9a-f]{4}-[1-5][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$' or jsonb_typeof(block->'alt') is distinct from 'string' or char_length(trim(block->>'alt')) not between 3 and 250 then return false; end if;
    else return false;
    end if;
  end loop;
  return true;
end; $$;

create table if not exists public.solar_articles (
  id uuid primary key default gen_random_uuid(),
  title text not null check (char_length(trim(title)) between 3 and 180),
  slug text not null unique check (char_length(slug) between 3 and 100 and slug ~ '^[a-z0-9]+(-[a-z0-9]+)*$'),
  description text not null default '' check (char_length(description) <= 320),
  category text not null check (category in ('Vodiči','Poticaji','Savjeti','Novosti')),
  cover_image_id uuid references public.solar_media(id) on delete restrict,
  cover_alt text not null default '' check (char_length(cover_alt) <= 250),
  blocks jsonb not null default '[]'::jsonb check (public.solar_valid_blocks(blocks)),
  status text not null default 'draft' check (status in ('draft','published')),
  author_id uuid not null default auth.uid() references auth.users(id),
  published_at timestamptz,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  constraint solar_published_content check (status <> 'published' or (cover_image_id is not null and char_length(trim(cover_alt)) >= 3 and char_length(trim(description)) >= 20 and published_at is not null and blocks @> '[{"type":"text"}]'::jsonb))
);
create index if not exists solar_articles_published_idx on public.solar_articles (published_at desc) where status = 'published';

create or replace function public.solar_touch_updated_at()
returns trigger language plpgsql set search_path = '' as $$ begin new.updated_at = now(); return new; end; $$;
drop trigger if exists solar_articles_touch on public.solar_articles;
create trigger solar_articles_touch before update on public.solar_articles for each row execute function public.solar_touch_updated_at();

alter table public.solar_articles enable row level security;
alter table public.solar_media enable row level security;
revoke all on public.solar_articles, public.solar_media from anon, authenticated;
grant select on public.solar_articles, public.solar_media to anon;
grant select, insert, update on public.solar_articles to authenticated;
grant select, insert on public.solar_media to authenticated;

drop policy if exists solar_articles_read on public.solar_articles;
create policy solar_articles_read on public.solar_articles for select to anon, authenticated using (status = 'published' or (select public.is_solar_admin()));
drop policy if exists solar_articles_insert on public.solar_articles;
create policy solar_articles_insert on public.solar_articles for insert to authenticated with check ((select public.is_solar_admin()) and author_id = (select auth.uid()));
drop policy if exists solar_articles_update on public.solar_articles;
create policy solar_articles_update on public.solar_articles for update to authenticated using ((select public.is_solar_admin())) with check ((select public.is_solar_admin()));

drop policy if exists solar_media_read on public.solar_media;
create policy solar_media_read on public.solar_media for select to anon, authenticated using (
  (select public.is_solar_admin()) or exists (
    select 1 from public.solar_articles a where a.status = 'published' and
    (a.cover_image_id = solar_media.id or a.blocks @> jsonb_build_array(jsonb_build_object('type','image','mediaId',solar_media.id::text)))
  )
);
drop policy if exists solar_media_insert on public.solar_media;
create policy solar_media_insert on public.solar_media for insert to authenticated with check ((select public.is_solar_admin()) and owner_id = (select auth.uid()));

insert into storage.buckets (id,name,public,file_size_limit,allowed_mime_types)
values ('solar-articles','solar-articles',true,8388608,array['image/webp'])
on conflict (id) do update set public=excluded.public,file_size_limit=excluded.file_size_limit,allowed_mime_types=excluded.allowed_mime_types;

drop policy if exists solar_storage_insert on storage.objects;
create policy solar_storage_insert on storage.objects for insert to authenticated with check (
  bucket_id = 'solar-articles' and (select public.is_solar_admin()) and (storage.foldername(name))[1] = (select auth.uid())::text
);
drop policy if exists solar_storage_admin_read on storage.objects;
create policy solar_storage_admin_read on storage.objects for select to authenticated using (bucket_id = 'solar-articles' and (select public.is_solar_admin()));
drop policy if exists solar_storage_cleanup on storage.objects;
create policy solar_storage_cleanup on storage.objects for delete to authenticated using (
  bucket_id = 'solar-articles' and (select public.is_solar_admin()) and (storage.foldername(name))[1] = (select auth.uid())::text
);

-- SOLARNI PORTAL: BRISANJE CLANAKA I SLIKE DO 100 KB
-- U postojecem projektu pokrenite CIJELU datoteku u Supabase SQL Editoru.
-- Sigurno je ponoviti. Ne brise postojece clanke ni slike.
-- Postojece slike ne komprimira retroaktivno; nove datoteke imaju limit 100000 bajtova.
begin;

alter table public.solar_articles add column if not exists attached_media_ids uuid[] not null default '{}';

-- Povijest spremljenih priloga omogucuje brisanje i zamijenjenih slika.
create or replace function public.solar_track_article_media()
returns trigger language plpgsql security definer set search_path = '' as $$
declare ids uuid[];
begin
  lock table public.solar_media in share row exclusive mode;
  select coalesce(array_agg(distinct id), '{}'::uuid[]) into ids from (
    select new.cover_image_id as id
    union select (b->>'mediaId')::uuid from jsonb_array_elements(new.blocks) b where b->>'type'='image'
  ) refs where id is not null;
  if exists(select 1 from unnest(ids) r(id) where not exists(select 1 from public.solar_media m where m.id=r.id)) then
    raise exception 'Slika vise ne postoji. Ponovno je prenesite.' using errcode='23503';
  end if;
  select coalesce(array_agg(distinct id), '{}'::uuid[]) into new.attached_media_ids
  from unnest(ids || case when tg_op='UPDATE' then old.attached_media_ids else '{}'::uuid[] end) r(id);
  return new;
end; $$;
revoke all on function public.solar_track_article_media() from public;
drop trigger if exists solar_articles_track_media on public.solar_articles;
create trigger solar_articles_track_media before insert or update on public.solar_articles
for each row execute function public.solar_track_article_media();
-- Backfill without changing publication/edit timestamps.
alter table public.solar_articles disable trigger solar_articles_touch;
update public.solar_articles set attached_media_ids=attached_media_ids;
alter table public.solar_articles enable trigger solar_articles_touch;

create table if not exists public.solar_storage_cleanup_queue (
  path text primary key,
  article_id uuid not null,
  created_at timestamptz not null default now()
);
alter table public.solar_storage_cleanup_queue enable row level security;
revoke all on public.solar_storage_cleanup_queue from anon, authenticated;
grant select,delete on public.solar_storage_cleanup_queue to authenticated;
drop policy if exists solar_cleanup_admin on public.solar_storage_cleanup_queue;
create policy solar_cleanup_admin on public.solar_storage_cleanup_queue for all to authenticated
using ((select public.is_solar_admin())) with check ((select public.is_solar_admin()));

-- Database changes and durable cleanup jobs are committed together.
-- Actual Storage files MUST be removed by the Storage API, not SQL.
create or replace function public.solar_delete_article(p_id uuid,p_updated_at timestamptz)
returns boolean language plpgsql security definer set search_path = '' as $$
declare target public.solar_articles%rowtype; asset public.solar_media%rowtype; object_path text;
begin
  if not public.is_solar_admin() then raise exception 'Administrator required' using errcode='42501'; end if;
  lock table public.solar_articles in share row exclusive mode;
  lock table public.solar_media in share row exclusive mode;
  select * into target from public.solar_articles where id=p_id;
  if not found then return false; end if;
  if p_updated_at is null or target.updated_at<>p_updated_at then
    raise exception 'Clanak je izmijenjen. Osvjezite stranicu prije brisanja.' using errcode='40001';
  end if;
  delete from public.solar_articles where id=p_id;
  for asset in select * from public.solar_media m where m.id=any(target.attached_media_ids)
    and not exists(select 1 from public.solar_articles a where m.id=any(a.attached_media_ids))
  loop
    for object_path in select v->>'path' from jsonb_array_elements(asset.variants) v loop
      if object_path not like asset.owner_id::text || '/' || asset.id::text || '/%' or object_path like '%..%' then
        raise exception 'Neispravna putanja slike';
      end if;
      insert into public.solar_storage_cleanup_queue(path,article_id) values(object_path,p_id) on conflict do nothing;
    end loop;
    delete from public.solar_media where id=asset.id;
  end loop;
  return true;
end; $$;
revoke all on function public.solar_delete_article(uuid,timestamptz) from public;
grant execute on function public.solar_delete_article(uuid,timestamptz) to authenticated;

-- Any administrator may complete queued cleanup, including another editor's uploads.
drop policy if exists solar_storage_cleanup on storage.objects;
create policy solar_storage_cleanup on storage.objects for delete to authenticated using (
  bucket_id='solar-articles' and (select public.is_solar_admin()) and (
    exists(select 1 from public.solar_storage_cleanup_queue q where q.path=name)
    or ((storage.foldername(name))[1]=(select auth.uid())::text and not exists(
      select 1 from public.solar_media m where m.variants @> jsonb_build_array(jsonb_build_object('path',name))
    ))
  )
);
update storage.buckets set file_size_limit=100000 where id='solar-articles';
commit;
